```python
def strategy_soros_forex_fundamental_break(df, asset):
    """
    George Soros: Break the Bank
    
    For: GBPUSD, EURUSD, USDJPY
    
    Rules:
    1. Identify overextended currencies
    2. Wait for policy divergence
    3. Build position on weakness
    4. Exit on capitulation
    """
    
    signals = pd.Series(0, index=df.index)
    
    # 1. Currency strength index (relative to 200 EMA)
    deviation_from_200 = (df['close'] - df['EMA_200']) / df['EMA_200'] * 100
    
    # 2. Overextended detection
    overvalued = deviation_from_200 > 8  # 8% above 200 EMA
    undervalued = deviation_from_200 < -8
    
    # 3. Policy catalyst (use RSI as proxy for momentum shift)
    catalyst = df['RSI'].diff(5) < -15  # Rapid momentum loss
    
    # 4. Entry conditions
    short_setup = overvalued & catalyst & (df['MACD'] < df['MACD_signal'])
    long_setup = undervalued & (-catalyst) & (df['MACD'] > df['MACD_signal'])
    
    # 5. Build position
    position = 0
    for i in range(1, len(df)):
        if position == 0:
            if short_setup.iloc[i]:
                position = -1
            elif long_setup.iloc[i]:
                position = 1
        
        # Exit on 200 EMA reclaim
        elif position == -1 and df['close'].iloc[i] < df['EMA_200'].iloc[i]:
            position = 0
        elif position == 1 and df['close'].iloc[i] > df['EMA_200'].iloc[i]:
            position = 0
        
        signals.iloc[i] = position
    
    return signals
```

### **2. LARRY WILLIAMS STRATEGY - FOR COMMODITIES**

def strategy_larry_williams_cot_seasonal(df, asset):
    """
    Larry Williams: COT + Seasonality
    
    For: GOLD, SILVER, CRUDE
    11,376% return in 12 months (1987)
    
    Rules:
    1. COT positioning (commercials vs specs)
    2. Seasonal patterns
    3. Cycle analysis
    4. Risk 4% per trade
    """
    
    signals = pd.Series(0, index=df.index)
    
    # 1. Seasonality (Gold example)
    if asset == 'GOLD':
        # Gold strong: Aug-Feb, weak: Mar-Jul
        month = df.index.month if isinstance(df.index, pd.DatetimeIndex) else 8
        bullish_season = (month >= 8) | (month <= 2)
        bearish_season = (month >= 3) & (month <= 7)
    
    # 2. COT proxy: Volume + Open Interest divergence
    vol_surge = df['volume'] > df['volume'].rolling(20).mean() * 1.5 if 'volume' in df.columns else True
    
    # 3. Cycle low/high (use RSI extremes as proxy)
    cycle_low = df['RSI'] < 35
    cycle_high = df['RSI'] > 65
    
    # 4. Trend confirmation
    uptrend = (df['EMA_20'] > df['EMA_50']) & (df['EMA_50'] > df['EMA_200'])
    downtrend = (df['EMA_20'] < df['EMA_50']) & (df['EMA_50'] < df['EMA_200'])
    
    # 5. Entry
    long_entry = bullish_season & cycle_low & uptrend & vol_surge
    short_entry = bearish_season & cycle_high & downtrend
    
    # 6. Exit (Williams: trail with 1.5× ATR)
    position = 0
    entry_price = 0
    
    for i in range(1, len(df)):
        if position == 0:
            if long_entry.iloc[i]:
                position = 1
                entry_price = df['close'].iloc[i]
            elif short_entry.iloc[i]:
                position = -1
                entry_price = df['close'].iloc[i]
        
        # Trailing stop: 1.5× ATR
        elif position == 1:
            stop = entry_price - (1.5 * df['ATR'].iloc[i])
            if df['close'].iloc[i] < stop:
                position = 0
        
        elif position == -1:
            stop = entry_price + (1.5 * df['ATR'].iloc[i])
            if df['close'].iloc[i] > stop:
                position = 0
        
        signals.iloc[i] = position
    
    return signals

def strategy_rakesh_jhunjhunwala_contrarian_conviction(df, asset):

    oversold = df['RSI'] < 35
    
    panic = sharp_drop & volume_surge & oversold
    
    # 2. GREED DETECTION (sell opportunity)
    sharp_rally = df['close'].pct_change(5) > 0.10
    overbought = df['RSI'] > 70
    
    greed = sharp_rally & overbought
    
    # 3. TREND CONFIRMATION (Jhunjhunwala: "Trend is your friend")
    long_term_uptrend = df['close'] > df['EMA_200']
    long_term_downtrend = df['close'] < df['EMA_200']

    buy_the_panic = panic & long_term_uptrend
    sell_the_greed = greed & long_term_downtrend
    
    # 5. POSITION MANAGEMENT
    # Jhunjhunwala: Exit losers ruthlessly, let winners run
    position = 0
    entry_price = 0
    bars_in_trade = 0
    
    for i in range(1, len(df)):
        bars_in_trade += 1
        
        if position == 0:
            if buy_the_panic.iloc[i]:
                position = 1
                entry_price = df['close'].iloc[i]
                bars_in_trade = 0
            elif sell_the_greed.iloc[i]:
                position = -1
                entry_price = df['close'].iloc[i]
                bars_in_trade = 0
        
        # EXIT LOSERS FAST (2% stop loss)
        elif position == 1:
            loss_pct = (df['close'].iloc[i] - entry_price) / entry_price
            if loss_pct < -0.02:
                position = 0  # Cut loss immediately
            # Exit on lower high after 20+ bars
            elif bars_in_trade > 20 and df['close'].iloc[i] < df['EMA_50'].iloc[i]:
                position = 0
        
        elif position == -1:
            loss_pct = (entry_price - df['close'].iloc[i]) / entry_price
            if loss_pct < -0.02:
                position = 0
            elif bars_in_trade > 20 and df['close'].iloc[i] > df['EMA_50'].iloc[i]:
                position = 0
        
        signals.iloc[i] = position
    
    return signals
